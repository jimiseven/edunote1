-- Migracion: tabla estable para la carga masiva de responsables por estudiante
-- Evita que los datos de la grilla se mezclen por relaciones o responsables compartidos.

CREATE TABLE IF NOT EXISTS `estudiantes_responsables_carga` (
  `id_estudiante` INT(11) NOT NULL,
  `padre_nombre` VARCHAR(255) DEFAULT NULL,
  `padre_telefono` VARCHAR(80) DEFAULT NULL,
  `madre_nombre` VARCHAR(255) DEFAULT NULL,
  `madre_telefono` VARCHAR(80) DEFAULT NULL,
  `actualizado_en` TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  PRIMARY KEY (`id_estudiante`),
  CONSTRAINT `fk_erc_estudiante`
    FOREIGN KEY (`id_estudiante`) REFERENCES `estudiantes` (`id_estudiante`)
    ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_general_ci;

INSERT INTO `estudiantes_responsables_carga` (`id_estudiante`, `padre_nombre`, `padre_telefono`, `madre_nombre`, `madre_telefono`)
SELECT
  er.`id_estudiante`,
  MAX(CASE WHEN er.`tipo_responsable` = 'PADRE' THEN COALESCE(er.`nombre_responsable`, TRIM(CONCAT(r.`nombre`, ' ', r.`apellido`))) END) AS padre_nombre,
  MAX(CASE WHEN er.`tipo_responsable` = 'PADRE' THEN COALESCE(er.`telefono_responsable`, r.`telefono`) END) AS padre_telefono,
  MAX(CASE WHEN er.`tipo_responsable` = 'MADRE' THEN COALESCE(er.`nombre_responsable`, TRIM(CONCAT(r.`nombre`, ' ', r.`apellido`))) END) AS madre_nombre,
  MAX(CASE WHEN er.`tipo_responsable` = 'MADRE' THEN COALESCE(er.`telefono_responsable`, r.`telefono`) END) AS madre_telefono
FROM `estudiantes_responsables` er
INNER JOIN `responsables` r ON r.`id_responsable` = er.`id_responsable`
GROUP BY er.`id_estudiante`
ON DUPLICATE KEY UPDATE
  `padre_nombre` = VALUES(`padre_nombre`),
  `padre_telefono` = VALUES(`padre_telefono`),
  `madre_nombre` = VALUES(`madre_nombre`),
  `madre_telefono` = VALUES(`madre_telefono`);
