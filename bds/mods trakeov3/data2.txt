-- Soporte de doble registro diario (manana/tarde) por estudiante
-- Seguro para ejecutar una sola vez en MariaDB/MySQL

START TRANSACTION;

-- 1) Permitir 2 marcas por dia (una por turno)
--    Reemplaza UNIQUE (id_estudiante, fecha) por UNIQUE (id_estudiante, fecha, turno)

ALTER TABLE asistencia
  DROP INDEX uk_estudiante_fecha;

ALTER TABLE asistencia
  ADD COLUMN turno ENUM('MANANA','TARDE') NOT NULL DEFAULT 'MANANA' AFTER fecha;

ALTER TABLE asistencia
  ADD UNIQUE KEY uk_estudiante_fecha_turno (id_estudiante, fecha, turno);

ALTER TABLE asistencia
  ADD KEY idx_fecha_turno (fecha, turno);

-- 2) Configuracion por curso de si requiere doble ingreso
CREATE TABLE IF NOT EXISTS asistencia_cursos_turnos (
  id_curso_turno INT(11) NOT NULL AUTO_INCREMENT,
  id_curso INT(11) NOT NULL,
  doble_turno TINYINT(1) NOT NULL DEFAULT 0 COMMENT '0=una marca diaria, 1=manana+tarde',
  estado TINYINT(1) NOT NULL DEFAULT 1,
  creado_por INT(11) DEFAULT NULL,
  created_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
  updated_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  PRIMARY KEY (id_curso_turno),
  UNIQUE KEY uk_curso_turno (id_curso),
  KEY idx_doble_turno_estado (doble_turno, estado),
  CONSTRAINT fk_asistencia_cursos_turnos_curso
    FOREIGN KEY (id_curso) REFERENCES cursos (id_curso)
    ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_general_ci;

-- 3) Horarios por curso y turno (manana/tarde), por rango de fechas
CREATE TABLE IF NOT EXISTS asistencia_horarios_curso_turno (
  id_horario_curso INT(11) NOT NULL AUTO_INCREMENT,
  id_curso INT(11) NOT NULL,
  fecha_inicio DATE NOT NULL,
  fecha_fin DATE NOT NULL,
  turno ENUM('MANANA','TARDE') NOT NULL,
  hora_ingreso TIME NOT NULL,
  tolerancia_min INT(11) NOT NULL DEFAULT 0,
  estado TINYINT(1) NOT NULL DEFAULT 1,
  creado_por INT(11) DEFAULT NULL,
  created_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (id_horario_curso),
  KEY idx_curso_rango_estado (id_curso, fecha_inicio, fecha_fin, estado),
  KEY idx_turno_estado (turno, estado),
  CONSTRAINT fk_asistencia_horarios_curso_turno_curso
    FOREIGN KEY (id_curso) REFERENCES cursos (id_curso)
    ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_general_ci;

-- 4) Backfill: todo registro historico queda como MANANA
UPDATE asistencia
SET turno = 'MANANA'
WHERE turno IS NULL OR turno = '';

COMMIT;
